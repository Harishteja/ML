# Linear regression with Gradient Descent
# Import libraries
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
import statsmodels.api as sm

np.random.seed(1)
num_samples = 100

X = np.random.rand(num_samples)
Y = 5 * X + 0.3 * np.random.randn(num_samples)  # Noise added

class LinearRegression_GD:

  def __init__(self,x,y,learning_rate,epoch):
    self.samples = x.size
    self.x = np.column_stack((np.ones_like(x),x))
    self.y = y
    self.learning_rate = learning_rate
    self.epoch = epoch

    self.samples, self.features = self.x.shape
    self.w = np.zeros(self.features)
    self.gradient = np.zeros(self.features)
  def compute_loss(self):
    error = self.y - self.y_hat
    loss = (1/self.samples) * np.dot(error.T,error)
    return loss
  def compute_gradient(self,x):
    return -(2/self.samples) *sum(np.dot(x.T,(self.y-self.y_hat)))
  def train(self):
    self.loss_history=[]
    for i in range(self.epoch):
      self.y_hat = np.dot(self.x,self.w)
      self.loss_history.append(self.compute_loss())
      for j in range(self.features):
        self.gradient[j] = self.compute_gradient(self.x[:,j].reshape(-1,1))

      self.w = self.w - self.learning_rate * self.gradient
    return self.w
  def predict(self,x):
    return np.dot(self.x,self.w)
  #intialize values
learning_rate = 0.2
epoch = 100

model = LinearRegression_GD(X,Y,learning_rate,epoch)
weights = model.train()
print("w =", weights)
Y_HAT = model.predict(X)
# Plot the regression line
fig= plt.figure(figsize=(10,6))
plt.title("Linear Regression")
plt.xlabel("Feature")
plt.ylabel("Output")
plt.plot(X,Y, marker='o', linestyle='none')
plt.plot(X,Y_HAT)
plt.show()
